# RPi.GPIO does NOT work on Pi 5 but some deps still import it at module level.

adafruit-platformdetect
adafruit-circuitpython-pca9685
adafruit-circuitpython-neopixel

gpiod                            # GPIO for LCD DC/RST pins (Pi 5 RP1 native)
